def _hull(name):
    """The applied Root Canal border (grace 5%) as triangles in (log10 amplitude, half-width, fast power)."""
    import json
    b = json.load(open(os.path.join(HERE, "rc_borders.json"), encoding="utf-8"))["halo"]["borders"][name]["border"]
    sh = [lv for lv in b["levels"] if abs(lv["grace"] - 0.05) < 1e-9][0]["shape"]
    V = np.asarray(sh["vertices"], float)          # amp, hf, hw
    P = np.column_stack([np.log10(V[:, 0]), V[:, 2], V[:, 1]])
    return [P[list(f)] for f in sh["faces"]]


def scatter3d(fig, rect, e, title, view=(20, -58), hulls=True, big=False):
    from mpl_toolkits.mplot3d.art3d import Poly3DCollection
    W, H = fig.get_size_inches()
    x, y, w, h = rect
    ax = fig.add_axes([x / W, 1 - (y + h) / H, w / W, h / H], projection="3d")
    ax.set_proj_type("persp", focal_length=0.6)
    zlo, zhi = -12, 50
    if hulls:
        for name, col in (("ds", F.C_DS), ("pspike", F.C_IED)):
            pc = Poly3DCollection(_hull(name), facecolor=col, edgecolor=col, linewidths=0.15, alpha=0.07)
            ax.add_collection3d(pc)
    for kind, s, al in (("DS", 1.6 if not big else 2.2, 0.22), ("far", 5, 0.8), ("TZ", 6 if not big else 9, 0.95), ("IED", 6 if not big else 9, 0.95)):
        q = e[e.kind == kind]
        q = q[np.isfinite(q.amp) & np.isfinite(q.hw) & np.isfinite(q.hf) & (q.hw <= 30)]
        ax.scatter(np.log10(q.amp), q.hw, q.hf, s=s, c=F.KIND_COLOR[kind], alpha=al, linewidths=0, depthshade=False, rasterized=True)
        if kind in ("TZ", "IED"):
            ax.scatter(np.log10(q.amp), q.hw, np.full(len(q), zlo), s=2, c=F.KIND_COLOR[kind], alpha=0.25, linewidths=0, depthshade=False,
                       rasterized=True)
    ax.set_xlim(2.0, 3.7)
    ax.set_ylim(2, 30)
    ax.set_zlim(zlo, zhi)
    ax.set_xticks([2, 2.5, 3, 3.5])
    ax.set_xticklabels(["100", "316", "1000", "3162"], fontsize=5.2)
    ax.set_yticks([5, 10, 20, 30])
    ax.set_zticks([-10, 0, 10, 20, 30, 40, 50])
    ax.tick_params(labelsize=5.2, pad=-1)
    ax.set_xlabel("Largest amplitude (µV)", fontsize=6 if big else 5.4, labelpad=-4)
    ax.set_ylabel("Half-width (ms)", fontsize=6 if big else 5.4, labelpad=-4)
    ax.set_zlabel("Fast power (dB)", fontsize=6 if big else 5.4, labelpad=-5)
    for axis in (ax.xaxis, ax.yaxis, ax.zaxis):
        axis.set_pane_color((0.97, 0.97, 0.98, 1.0))
        axis._axinfo["grid"]["color"] = (0.88, 0.89, 0.91, 1)
        axis._axinfo["grid"]["linewidth"] = 0.4
    ax.view_init(elev=view[0], azim=view[1])
    ax.set_title(title, fontsize=6.5 if big else 6, loc="left", pad=0)
    return ax
