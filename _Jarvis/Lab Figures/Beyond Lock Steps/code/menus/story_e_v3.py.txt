def _hull_xy(name, i, j):
    """The border's outline in one projection: the convex hull of its vertices on two of the three numbers
    (columns: 0 = log10 amplitude, 1 = half-width, 2 = fast power)."""
    from scipy.spatial import ConvexHull
    tri = _hull(name)
    P = np.unique(np.vstack(tri), axis=0)[:, [i, j]]
    hh = ConvexHull(P)
    return P[hh.vertices]
